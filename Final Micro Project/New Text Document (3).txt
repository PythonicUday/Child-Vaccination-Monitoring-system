// Total registered children
$result = mysqli_query($conn,
    "SELECT COUNT(*) AS total FROM children");
$totalChildren = mysqli_fetch_assoc($result)['total'];

// Completed vaccinations
$result = mysqli_query($conn,
    "SELECT COUNT(*) AS total
     FROM vaccination_records
     WHERE status = 'completed'");
$completed = mysqli_fetch_assoc($result)['total'];

// Pending vaccinations
$result = mysqli_query($conn,
    "SELECT COUNT(*) AS total
     FROM vaccination_records
     WHERE status = 'pending'");
$upcoming = mysqli_fetch_assoc($result)['total'];

// Recent activity
$recentActivity = mysqli_query($conn,
    "SELECT ch.name AS child_name,
            v.vaccine_name,
            vr.status,
            vr.vaccination_date
     FROM vaccination_records vr
     JOIN children ch ON vr.child_id = ch.child_id
     JOIN vaccines v ON vr.vaccine_id = v.vaccine_id
     ORDER BY vr.record_id DESC
     LIMIT 5");
